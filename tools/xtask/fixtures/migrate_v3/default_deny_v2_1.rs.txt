//! Default-deny receiver identity (second post-fix review of F03, V3-O04):
//! migrate-v3 input.
//!
//! Every call marked `unrelated` returns 1 (or its original text) in v2.1, and
//! its unrelated target also offers `replace_*` returning 2, so a wrong rename
//! would still compile and silently change the result.
//! `tools/xtask/src/migration/tests/default_deny.rs` migrates this file as a
//! module of a crate and asserts that the result is byte-for-byte
//! `tools/xtask/tests/migrated_v3_identity/default_deny.rs`, which
//! `tools/xtask/tests/migrated_v3.rs` compiles through the real v3
//! `concord::api!` and executes. The root of that test target stands in for
//! the third-party crate `extpatch` (`extern crate self as extpatch;`).

use concord::endpoint::RequestPlan;
use concord::{ApiError, Empty, api};

api! {
    client Api {
        base "https://review.example.test/";
        timeout 5s;
        values {
            locale?: String;
        }
        query {
            "locale" = client.locale;
        }
        GET item(x?: u32) -> () via Empty {
            path ["item"];
            query { "x" = endpoint.x; }
        }
    }
}

/// Unrelated items whose names collide with the generated ones.
pub mod other {
    /// The generated request's leaf name, with both setter names.
    pub struct ItemRequest;

    impl ItemRequest {
        /// The original call's result.
        pub fn set_x(self, _: Option<u32>) -> u32 {
            1
        }

        /// What a wrong rename would return instead.
        pub fn replace_x(self, _: Option<u32>) -> u32 {
            2
        }
    }
}

// ----- (1) declarations gated by `#[cfg]` ------------------------------------

/// The shadowing `let` is compiled in no configuration.
pub fn cfg_let(r: other::ItemRequest, g: api::ItemRequest) -> u32 {
    let _keep = g;
    #[cfg(any())]
    let r = _keep;
    r.set_x(None) // unrelated: cfg let
}

/// The explicit import over the glob exists only for another target.
pub mod cfg_import {
    use super::other::*;
    #[cfg(target_os = "none")]
    use super::api::ItemRequest;

    pub fn call(r: ItemRequest) -> u32 {
        r.set_x(None) // unrelated: cfg import
    }
}

// ----- (2) arguments of macros that do not evaluate them in place ------------

/// Rebinds the name it is given before evaluating the expression.
macro_rules! with_other {
    ($name:ident, $call:expr) => {{
        let $name = other::ItemRequest;
        $call
    }};
}

pub fn macro_rebinds_argument(r: api::ItemRequest) -> u32 {
    let _keep = r;
    with_other!(r, r.set_x(None)) // unrelated: macro rebinds argument
}

/// Calls the method tokens on another receiver.
macro_rules! on_other {
    ($receiver:ident . $($call:tt)*) => {{
        let _ = &$receiver;
        other::ItemRequest.$($call)*
    }};
}

pub fn macro_restructures_tokens(r: api::ItemRequest) -> u32 {
    on_other!(r.set_x(None)) // unrelated: macro restructures tokens
}

pub fn stringify_argument(r: api::ItemRequest) -> &'static str {
    let _ = r;
    stringify!(r.set_x(None)) // unrelated: stringify
}

pub fn stringify_builder() -> &'static str {
    stringify!(ApiBuilder) // unrelated: stringify builder
}

// ----- (3) traits of another crate --------------------------------------------

/// A blanket extension trait of another crate intercepts a `Box` receiver.
pub mod ext_blanket {
    use extpatch::Patch;

    pub fn call(b: Box<super::api::ItemRequest>) -> u32 {
        b.set_x(None) // unrelated: external blanket trait
    }
}

/// Another crate's trait whose methods are provided by default.
pub mod ext_default {
    use extpatch::Defaulted;

    impl Defaulted for Box<super::api::ItemRequest> {}

    pub fn call(b: Box<super::api::ItemRequest>) -> u32 {
        extpatch::touch();
        b.set_x(None) // unrelated: external default method
    }
}

// ----- (4) trait implementations generated by a crate-local macro ------------

/// Implemented for `Box<api::ItemRequest>` by `patch_for!` only.
pub trait MacroPatch {
    /// The original call's result.
    fn set_x(self, value: Option<u32>) -> u32;
    /// What a wrong rename would return instead.
    fn replace_x(self, value: Option<u32>) -> u32;
}

macro_rules! patch_for {
    ($target:ty) => {
        impl super::MacroPatch for $target {
            fn set_x(self, _: Option<u32>) -> u32 {
                1
            }
            fn replace_x(self, _: Option<u32>) -> u32 {
                2
            }
        }
    };
}

mod impls {
    patch_for!(Box<super::api::ItemRequest>);
}

pub fn macro_impl_box(b: Box<api::ItemRequest>) -> u32 {
    b.set_x(None) // unrelated: macro-generated impl on Box
}

/// Implemented for the generated client by `nav_for!` only: its by-value
/// `item` is found before the generated `&self` navigation.
pub trait Nav {
    /// The unrelated navigation.
    fn item(self) -> other::ItemRequest;
}

macro_rules! nav_for {
    ($target:ty) => {
        impl super::Nav for $target {
            fn item(self) -> super::other::ItemRequest {
                super::other::ItemRequest
            }
        }
    };
}

mod nav_impls {
    nav_for!(super::Api);
}

pub fn macro_impl_navigation(client: Api) -> u32 {
    client.item().set_x(None) // unrelated: macro-generated by-value navigation
}

// ----- raw identifiers ----------------------------------------------------------

pub fn raw_foreign(r: other::ItemRequest) -> u32 {
    r.r#set_x(None) // unrelated: raw identifier
}

// ----- genuine generated calls ------------------------------------------------

/// Genuine generated calls in v2.1 exact outer-slot replacement: receivers
/// that reach the generated method at the first probe step (an exact request,
/// navigation through `&Api`), a binding under a trivially true `#[cfg]`, a
/// raw identifier and the arguments of standard expression macros.
pub fn genuine_plans(client: &Api) -> Vec<Result<RequestPlan, ApiError>> {
    #[cfg(all())]
    let gated = client.item();
    let gated = gated.set_x(Some(11)); // genuine: trivially true cfg
    let exact = client.item().set_x(Some(12)); // genuine: reference to the client
    let raw = client.item().r#set_x(Some(13)); // genuine: raw identifier
    let listed = vec![client.item().set_x(Some(14))]; // genuine: vec! argument
    assert!(client.item().set_x(None).plan().is_ok()); // genuine: assert! argument
    let formatted = format!("{:?}", client.item().set_x(None).plan().is_ok()); // genuine: format! argument
    assert_eq!(formatted, "true");
    let mut plans = vec![gated.plan(), exact.plan(), raw.plan()];
    plans.extend(listed.into_iter().map(|request| request.plan()));
    plans
}
