//! Third post-fix review of F03 (round 4, V3-O04): migrate-v3 input.
//!
//! Every call marked `unrelated` returns 1 in v2.1, and its unrelated target
//! also offers `replace_x` returning 2, so a wrong rename would still compile
//! and silently change the result.
//! `tools/xtask/src/migration/tests/round4.rs` migrates this file as the module
//! `round4` of a crate (with `round4_files/local_data.rs` beside it and a
//! stray file at the module's default location) and asserts that the result
//! is byte-for-byte `tools/xtask/tests/migrated_v3_identity/round4.rs`, which
//! `tools/xtask/tests/migrated_v3.rs` compiles through the real v3
//! `concord::api!` and executes.

use concord::endpoint::RequestPlan;
use concord::{ApiError, Empty, api};

api! {
    client Api {
        base "https://review.example.test/";
        timeout 5s;
        values {
            locale?: String;
        }
        query {
            "locale" = client.locale;
        }
        GET item(x?: u32) -> () via Empty {
            path ["item"];
            query { "x" = endpoint.x; }
        }
    }
}

/// Unrelated items whose names collide with the generated ones.
pub mod other {
    /// The generated request's leaf name, with both setter names.
    #[derive(Clone, Copy)]
    pub struct ItemRequest;

    impl ItemRequest {
        /// The original call's result.
        pub fn set_x(self, _: Option<u32>) -> u32 {
            1
        }

        /// What a wrong rename would return instead.
        pub fn replace_x(self, _: Option<u32>) -> u32 {
            2
        }
    }
}

// ----- (D1) a `let` annotation coerces the initializer ------------------------

/// Navigation through a trait object of the client.
pub trait Nav {
    /// The unrelated navigation.
    fn item(&self) -> other::ItemRequest;
}

impl Nav for Api {
    fn item(&self) -> other::ItemRequest {
        other::ItemRequest
    }
}

pub fn dyn_ref_annotation(client: &Api) -> u32 {
    let r: &dyn Nav = client;
    r.item().set_x(None) // unrelated: dyn reference annotation
}

/// A trait object of the generated request.
pub trait DynExt {
    /// The original call's result.
    fn set_x(&self, value: Option<u32>) -> u32;
    /// What a wrong rename would return instead.
    fn replace_x(&self, value: Option<u32>) -> u32;
}

impl DynExt for api::ItemRequest {
    fn set_x(&self, _: Option<u32>) -> u32 {
        1
    }
    fn replace_x(&self, _: Option<u32>) -> u32 {
        2
    }
}

pub fn dyn_box_annotation(boxed: Box<api::ItemRequest>) -> u32 {
    let b: Box<dyn DynExt> = boxed;
    b.set_x(None) // unrelated: dyn box annotation
}

/// An alias of a trait object.
pub type DynNav = dyn Nav;

pub fn alias_of_dyn_annotation(client: &Api) -> u32 {
    let r: &DynNav = client;
    r.item().set_x(None) // unrelated: alias of dyn annotation
}

// ----- (D3) field visibility --------------------------------------------------------

/// A private field and an accessible field of the same name behind `Deref`.
pub mod holder {
    /// The `Deref` target.
    pub struct Holder {
        /// The accessible field.
        pub inner: super::other::ItemRequest,
    }

    /// Its generated-typed field is private.
    pub struct S {
        inner: super::api::ItemRequest,
        /// The `Deref` target.
        pub other: Holder,
    }

    impl std::ops::Deref for S {
        type Target = Holder;
        fn deref(&self) -> &Holder {
            &self.other
        }
    }

    impl S {
        pub fn new(inner: super::api::ItemRequest) -> Self {
            Self {
                inner,
                other: Holder {
                    inner: super::other::ItemRequest,
                },
            }
        }

        /// Inside its module the private field is the one accessed.
        pub fn own(self) -> Result<super::RequestPlan, super::ApiError> {
            self.inner.set_x(Some(23)).plan() // genuine: private field inside its module
        }
    }
}

pub fn private_field(s: holder::S) -> u32 {
    s.inner.set_x(None) // unrelated: private field
}

/// A `pub(super)` field seen from outside its parent.
pub mod outer {
    /// Declares the restricted field.
    pub mod fields {
        /// The `Deref` target.
        pub struct Open {
            /// The accessible field.
            pub a: super::super::other::ItemRequest,
        }

        /// Its generated-typed field is visible in `outer` only.
        pub struct V {
            pub(super) a: super::super::api::ItemRequest,
            /// The `Deref` target.
            pub open: Open,
        }

        impl std::ops::Deref for V {
            type Target = Open;
            fn deref(&self) -> &Open {
                &self.open
            }
        }

        impl V {
            pub fn new(a: super::super::api::ItemRequest) -> Self {
                Self {
                    a,
                    open: Open {
                        a: super::super::other::ItemRequest,
                    },
                }
            }
        }
    }
}

pub fn pub_super_field(v: outer::fields::V) -> u32 {
    v.a.set_x(None) // unrelated: pub(super) field
}

// ----- genuine generated calls ------------------------------------------------

pub fn sync_make(client: &Api) -> api::ItemRequest {
    client.item()
}

/// Genuine generated calls in v2.1 exact outer-slot replacement: an exact
/// annotation, a plain `_` annotation, a synchronous function result, a
/// private field inside its module and a type read through the `#[path]`
/// module.
pub fn genuine_plans(client: &Api) -> Vec<Result<RequestPlan, ApiError>> {
    let r: &Api = client;
    let annotated = r.item().set_x(Some(21)); // genuine: exact annotation
    let inferred: _ = client.item();
    let inferred = inferred.set_x(Some(22)); // genuine: plain infer annotation
    let synchronous = sync_make(client).set_x(Some(24)); // genuine: sync fn result
    let through_path: thread::local_data::G = client.item();
    let through_path = through_path.set_x(Some(25)); // genuine: path module type
    vec![
        annotated.plan(),
        inferred.plan(),
        holder::S::new(client.item()).own(),
        synchronous.plan(),
        through_path.plan(),
    ]
}
