// The v2.1 baseline declaration for `src/bin/head_cost.rs` (V3-O07).
//
// This is the v2.1-syntax equivalent of `declaration.rs`: the same two
// clients, base URLs, owners, routes, query/header program, credential
// placement and rate policy. For the baseline measurement it is copied to
// `tools/perf/src/head_cost/declaration.rs` inside a `git archive
// baseline-v2.1` export together with an unchanged copy of
// `tools/perf/src/bin/head_cost.rs` (see
// `tools/perf/measurements/head_cost_linux_2026-09-26.txt`). It is not
// compiled in the v3 workspace.

use concord::{Bytes, api};

api! {
    client HeadApi {
        base "https://head-cost.example.test/v1/";
        timeout 30s;

        values {
            tenant: String;
        }

        credentials {
            secret api_secret;
            credential application_key = bearer(secret.api_secret);
        }

        query {
            "tenant" = client.tenant;
        }

        headers {
            "X-Client-Version" = "3";
        }

        scope records(collection: String) {
            path ["records", records.collection];

            GET show(id: String) -> Bytes via Binary {
                path [endpoint.id];
            }

            GET search(
                term: String,
                count: u32 = 25,
                page?: u32,
                tags: Vec<String>,
            ) -> Bytes via Binary {
                path ["search"];
                query {
                    "term" = endpoint.term;
                    "count" = endpoint.count;
                    "page" = endpoint.page;
                    "tag" = each endpoint.tags;
                }
            }
        }

        GET protected() -> Bytes via Binary {
            path ["protected"];
            auth { bearer application_key; }
        }

        GET limited() -> Bytes via Binary {
            path ["limited"];
            rate_limit {
                id "concord.head_cost.1";
                key [origin];
                cost 1;
                allow 1000000 per 1s;
            }
        }
    }
}

api! {
    client ExecApi {
        base "http://head-cost.example.test/";
        timeout 30s;

        GET bounded() -> Bytes via Binary {
            path ["bounded"];
        }
    }
}
