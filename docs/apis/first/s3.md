## The description

The tree follows the path-style URL (`s3.{region}.amazonaws.com/{bucket}/{key}`). For AWS, the signing transport moves the bucket into the host as the request leaves (finding 2).

```rust
use concord::read::Stream;
use concord::{cursor, Body};
use crate::model::*;                    // the data types and small functions, beside the macro
use crate::xml::{xml, Xml, XmlReply};

concord::api! {
    /// An object store that speaks the S3 protocol: Amazon S3, or a store that copies it
    /// (MinIO, Ceph, Cloudflare R2, …).
    ///
    /// **Every request must be signed** (Signature Version 4). Make the client with
    /// [`S3::signed`] for AWS, or [`S3::compatible`] for another store. `S3::new` compiles,
    /// and gives a client that does not sign: S3 answers it `403`.
    pub client S3(
        /// The region, as AWS writes it: `eu-west-1`.
        region: Region,
    ) {
        base "https://s3.{region}.amazonaws.com";
        read Xml;
        problem S3Problem;

        /// How the object is stored and served back: on `put`, `copy_from`, `create_multipart_upload`.
        part object_headers(content_type: Option<String>, cache_control: Option<String>,
                            content_disposition: Option<String>, content_encoding: Option<String>,
                            storage_class: Option<StorageClass>) {
            headers {
                "content-type" = content_type;
                "cache-control" = cache_control;
                "content-disposition" = content_disposition;
                "content-encoding" = content_encoding;
                "x-amz-storage-class" = storage_class;
            }
        }

        /// ListBuckets: the buckets of the account. In pages since 2024 (unsure); a store
        /// that does not page sends one page with no token.
        GET list_buckets(prefix: Option<String>, bucket_region: Option<Region>,
                         max_buckets: Option<u32>, continuation_token: Option<String>) -> BucketList {
            query { prefix; "bucket-region" = bucket_region; "max-buckets" = max_buckets;
                    "continuation-token" = continuation_token; }
            paginate cursor(continuation_token, BucketList::next_token);
        }

        /// One bucket. Written in the path here; on AWS the signer puts it in the host
        /// (`{name}.s3.{region}.amazonaws.com`), unless the name has a dot. See [`Addressing`].
        scope bucket(name: String) {
            path "{name}";

            /// HeadBucket. `None` when there is no such bucket. One that exists and is not
            /// yours is a `403`, an error.
            HEAD head() -> Option<BucketHead> {
                status {
                    2xx { headers } => Some(BucketHead::from_headers(headers)),
                    301 { headers } => Some(BucketHead::from_headers(headers)),   // it lives in another region (unsure)
                    404 => None,
                }
            }

            /// CreateBucket, in the region of this client.
            PUT create(object_ownership: Option<ObjectOwnership>) {
                headers { "x-amz-object-ownership" = object_ownership; }
                body bucket_configuration(client.region);
            }

            /// DeleteBucket. The bucket must be empty.
            DELETE delete();

            /// ListObjectsV2: the objects in the order of their keys, a page at a time. With a
            /// `delimiter`, the keys that share a prefix up to it come once, in `common_prefixes`.
            GET list_objects(prefix: Option<String>, delimiter: Option<String>, max_keys: Option<u16>,
                             start_after: Option<String>, fetch_owner: Option<bool>,
                             continuation_token: Option<String>) -> ListBucketResult {
                query { "list-type" = "2"; prefix; delimiter; "max-keys" = max_keys; "start-after" = start_after;
                        "fetch-owner" = fetch_owner; "continuation-token" = continuation_token; }
                paginate |call, page| match (page.value().is_truncated, &page.value().next_continuation_token) {
                    (true, Some(token)) => Ok(Some(call.continuation_token(token.clone()))),
                    (true, None) => Err(concord::mismatch("IsTruncated without a NextContinuationToken")),
                    (false, _) => Ok(None),
                };
            }

            /// DeleteObjects: up to 1000 keys in one request. A key that could not be deleted is
            /// in `errors`, and the call itself succeeds.
            POST delete_objects(keys: Vec<String>, quiet: Option<bool>) -> DeleteResult {
                query { "delete" = ""; }
                headers { "content-md5" = DeleteRequest::new(keys, *quiet).content_md5(); }
                body xml(&DeleteRequest::new(keys, *quiet));
                idempotent;
            }

            /// ListMultipartUploads: uploads that were started and neither completed nor aborted.
            GET list_multipart_uploads(prefix: Option<String>, max_uploads: Option<u16>,
                                       key_marker: Option<String>,
                                       upload_id_marker: Option<UploadId>) -> ListMultipartUploadsResult {
                query { "uploads" = ""; prefix; "max-uploads" = max_uploads;
                        "key-marker" = key_marker; "upload-id-marker" = upload_id_marker; }
                paginate |call, page| page.value().next_markers()
                    .map(|(key, id)| call.key_marker(key).upload_id_marker(id));
            }

            /// One object. Its key may hold `/`: `photos/2026/cat.jpg`. A key that ends with
            /// `/`, or holds `//`, cannot be sent (finding 3).
            scope object(key: String) {
                path "{*key}";

                /// GetObject: the head, and the bytes as they come. With `range`, a part of them
                /// (`206`). `idle_timeout` goes on while you read.
                GET get(range: Option<ByteRange>, if_match: Option<ETag>, version_id: Option<VersionId>) -> Download {
                    query { "versionId" = version_id; }
                    headers { range; "if-match" = if_match; }
                    read Stream;
                    status { 2xx { body, headers } => Download::new(body, headers)? }
                }

                /// GetObject with `If-None-Match`: `None` while the object still has this ETag (`304`).
                GET get_if_changed(etag: ETag, range: Option<ByteRange>) -> Option<Download> {
                    headers { "if-none-match" = etag; range; }
                    read Stream;
                    status {
                        2xx { body, headers } => Some(Download::new(body, headers)?),
                        304 => None,
                    }
                }

                /// HeadObject: the head that `get` gives, with no bytes. `None` when there is no
                /// such key. (A caller without `s3:ListBucket` gets `403` instead: S3 hides which.)
                HEAD head(version_id: Option<VersionId>) -> Option<ObjectHead> {
                    query { "versionId" = version_id; }
                    status {
                        2xx { headers } => Some(ObjectHead::from_headers(headers)?),
                        404 => None,
                    }
                }

                /// PutObject: up to 5 GiB in one request. A stream is sent once, and needs
                /// `content_length`: S3 refuses a chunked body. `.if_none_match("*")` writes only
                /// if there is no object at this key (`412` otherwise). User metadata: `.meta(…)`.
                PUT put(content: Body, content_length: Option<u64>, if_none_match: Option<String>,
                        ..object_headers) -> Stored {
                    headers { "content-length" = content_length; "if-none-match" = if_none_match; }
                    body content;
                    status {
                        2xx { headers } => Stored {
                            etag: ETag::new(headers.require("etag")?),
                            version_id: headers.get("x-amz-version-id").map(VersionId::new),
                        },
                    }
                }

                /// CopyObject: this object, made on the server from `source` (5 GiB at most).
                /// A `200` can still carry an error; it is read, and is an error here too.
                PUT copy_from(source: CopySource, metadata_directive: Option<MetadataDirective>,
                              source_if_match: Option<ETag>, ..object_headers) -> CopyObjectResult {
                    headers {
                        "x-amz-copy-source" = source;
                        "x-amz-metadata-directive" = metadata_directive;
                        "x-amz-copy-source-if-match" = source_if_match;
                    }
                    status { 200 { body: reply } => XmlReply::into_result(reply)? }
                }

                /// DeleteObject. A key that is not there is "deleted" too: `204` either way.
                DELETE delete(version_id: Option<VersionId>) {
                    query { "versionId" = version_id; }
                }

                /// CreateMultipartUpload. Each call starts a new upload, kept (and billed) until it
                /// is completed or aborted; so it is never sent twice.
                POST create_multipart_upload(..object_headers) -> InitiateMultipartUploadResult {
                    query { "uploads" = ""; }
                }

                /// One multipart upload of this object.
                scope multipart(upload_id: UploadId) {
                    query { "uploadId" = upload_id; }

                    /// UploadPart: part `part_number` (1 to 10 000; 5 MiB at least, but the last).
                    /// Bytes held in memory can be sent again; a stream is sent once.
                    PUT upload_part(part_number: u16, part: Body, content_length: Option<u64>) -> CompletedPart {
                        query { "partNumber" = part_number; }
                        headers { "content-length" = content_length; }
                        body part;
                        status { 2xx { headers } => CompletedPart { part_number: *part_number, etag: ETag::new(headers.require("etag")?) } }
                    }

                    /// CompleteMultipartUpload: the object, made of these parts in this order. It
                    /// can take minutes (S3 sends spaces meanwhile), and a `200` can carry an error.
                    POST complete(parts: Vec<CompletedPart>) -> CompleteMultipartUploadResult {
                        body xml(&CompleteMultipartUpload { parts });
                        status { 200 { body: reply } => XmlReply::into_result(reply)? }
                        idempotent;     // the same parts make the same object (unsure: a repeat after success may say NoSuchUpload)
                    }

                    /// AbortMultipartUpload: the parts are dropped, and no longer billed.
                    DELETE abort();

                    /// ListParts: the parts uploaded so far.
                    GET list_parts(max_parts: Option<u16>, part_number_marker: Option<u16>) -> ListPartsResult {
                        query { "max-parts" = max_parts; "part-number-marker" = part_number_marker; }
                        paginate cursor(part_number_marker, ListPartsResult::next_marker);
                    }
                }
            }
        }
    }
}
```

### The functions used in arms, entries and bodies (`src/model.rs`)

```rust
/// The head of an object, from the headers of GetObject or HeadObject.
#[derive(Debug, Clone)]
pub struct ObjectHead {
    /// The length of what comes: of the part, after a `range`.
    pub content_length: u64,
    pub content_type: Option<String>,
    pub etag: ETag,
    pub last_modified: std::time::SystemTime,
    pub storage_class: Option<String>,                     // absent for `STANDARD`
    pub version_id: Option<VersionId>,
    /// `x-amz-meta-*`, with the prefix taken off. S3 keeps the names in lower case.
    pub metadata: BTreeMap<String, String>,
}

impl ObjectHead {
    /// In the arms of `get`, `get_if_changed`, `head`. A head that is not what S3 promises is a
    /// Mismatch: `concord::mismatch` gives a `concord::Fail` (sheet, item 11).
    pub fn from_headers(headers: &concord::Headers) -> Result<ObjectHead, concord::Fail<S3Problem>> {
        let length = headers.require("content-length")?;
        let modified = headers.require("last-modified")?;
        Ok(ObjectHead {
            content_length: length.parse().map_err(|_| concord::mismatch("content-length is not a number"))?,
            content_type: headers.get("content-type").map(String::from),
            etag: ETag::new(headers.require("etag")?),
            last_modified: httpdate::parse_http_date(modified).map_err(|_| concord::mismatch("last-modified is not a date"))?,
            storage_class: headers.get("x-amz-storage-class").map(String::from),
            version_id: headers.get("x-amz-version-id").map(VersionId::new),
            metadata: headers.iter()             // (unsure: the guide shows only `get` and `require`)
                .filter_map(|(name, value)| Some((name.strip_prefix("x-amz-meta-")?.to_owned(), value.to_owned())))
                .collect(),
        })
    }
}

/// What `get` gives: the head, and the bytes as they come. `content_range` on a `206`.
pub struct Download { pub head: ObjectHead, pub content_range: Option<String>, pub bytes: concord::ByteStream }

impl Download {
    pub fn new(bytes: concord::ByteStream, headers: &concord::Headers) -> Result<Download, concord::Fail<S3Problem>> {
        let content_range = headers.get("content-range").map(String::from);
        Ok(Download { head: ObjectHead::from_headers(headers)?, content_range, bytes })
    }
}

/// HeadBucket: where the bucket lives.
pub struct BucketHead { pub region: Option<Region> }
impl BucketHead {
    pub fn from_headers(headers: &concord::Headers) -> BucketHead {
        BucketHead { region: headers.get("x-amz-bucket-region").map(Region::new) }

/// `range`: `bytes=0-1023`, `bytes=1024-`, or the last n bytes, `bytes=-500`.
#[derive(Clone)]
pub enum ByteRange { Within(u64, u64), From(u64), Last(u64) }

impl Serialize for ByteRange {
    fn serialize<S: Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
        serializer.serialize_str(&match self {
            ByteRange::Within(first, last) => format!("bytes={first}-{last}"),
            ByteRange::From(first) => format!("bytes={first}-"),
            ByteRange::Last(count) => format!("bytes=-{count}"),
        })
    }
}

/// `x-amz-copy-source`: `{bucket}/{key}`, the key URL-encoded piece by piece, maybe `?versionId=`.
#[derive(Clone)]
pub struct CopySource { pub bucket: String, pub key: String, pub version_id: Option<VersionId> }

impl Serialize for CopySource {
    fn serialize<S: Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
        let key: Vec<String> = self.key.split('/').map(|piece| utf8_percent_encode(piece, STRICT).to_string()).collect();
        let version = self.version_id.as_ref().map(|v| format!("?versionId={}", v.as_str())).unwrap_or_default();
        serializer.serialize_str(&format!("{}/{}{version}", self.bucket, key.join("/")))
    }
}

/// The body of DeleteObjects.
#[derive(Serialize)]
#[serde(rename = "Delete")]
pub struct DeleteRequest<'a> {
    #[serde(rename = "Object")] objects: Vec<KeyOnly<'a>>,
    #[serde(rename = "Quiet", skip_serializing_if = "Option::is_none")] quiet: Option<bool>,
}

#[derive(Serialize)]
struct KeyOnly<'a> { #[serde(rename = "Key")] key: &'a str }

impl<'a> DeleteRequest<'a> {
    pub fn new(keys: &'a [String], quiet: Option<bool>) -> Self {
        DeleteRequest { objects: keys.iter().map(|key| KeyOnly { key: key.as_str() }).collect(), quiet }
    }

    /// The `Content-MD5` that S3 requires on DeleteObjects: the same XML, written a second
    /// time, and hashed. `None` if it cannot be written: then `body` fails too, and nothing is sent.
    pub fn content_md5(&self) -> Option<String> {
        let text = quick_xml::se::to_string(self).ok()?;
        Some(base64::engine::general_purpose::STANDARD.encode(md5::compute(text).0))
    }
}

/// The body of CompleteMultipartUpload.
#[derive(Serialize)]
#[serde(rename = "CompleteMultipartUpload")]
pub struct CompleteMultipartUpload<'a> { #[serde(rename = "Part")] pub parts: &'a [CompletedPart] }

// The pagers' small functions.
impl BucketList { pub fn next_token(&self) -> Option<String> { self.continuation_token.clone() } }
impl ListPartsResult {
    pub fn next_marker(&self) -> Option<u16> { self.next_part_number_marker.filter(|_| self.is_truncated) }
}
impl ListMultipartUploadsResult {
    /// Both markers move together.
    pub fn next_markers(&self) -> Option<(String, UploadId)> {
        if !self.is_truncated { return None; }
        Some((self.next_key_marker.clone()?, self.next_upload_id_marker.clone()?))
    }
}
```

### XML in and out (`src/xml.rs`)

```rust
use serde::{de::DeserializeOwned, Deserialize, Serialize};

/// `read Xml;`: the whole body as XML, into any type that `serde` can read. 10 MiB at most.
pub struct Xml;

impl<T: DeserializeOwned + Send> concord::Reader<T> for Xml {
    async fn read(&self, incoming: concord::read::Incoming) -> Result<T, concord::read::ReadError> {
        // (unsure) The guide names `Incoming` and `ReadError` and shows none of their methods.
        let bytes = incoming.whole(10 * concord::MIB).await?;
        quick_xml::de::from_reader(&bytes[..]).map_err(concord::read::ReadError::decode)
    }
}

/// `body xml(&value);`: the value as XML, labelled `application/xml`.
pub fn xml<T: Serialize>(value: &T) -> Result<concord::Body, quick_xml::SeError> {
    Ok(concord::Body::new(quick_xml::se::to_string(value)?, "application/xml"))
}

/// What CopyObject and CompleteMultipartUpload may answer with a `200`: the result, or an
/// `<Error>`. The root element picks the case (unsure: how quick-xml maps a root to a variant).
#[derive(Deserialize)]
pub enum XmlReply<T> { Error(S3Problem), #[serde(untagged)] Done(T) }

impl<T> XmlReply<T> {
    /// In an arm: `XmlReply::into_result(reply)?`. Its `Err` ends the call as a Refused error,
    /// with `problem()` set, as any other refusal of S3.
    pub fn into_result(self) -> Result<T, S3Problem> {
        match self { XmlReply::Done(value) => Ok(value), XmlReply::Error(problem) => Err(problem) }
    }
}
```

### Signature Version 4: the transport (`src/signing.rs`)

```rust
use std::time::Duration;
use chrono::{DateTime, Utc};
use concord::transport::{Transport, TransportError, TryInfo};
use hmac::{Hmac, Mac};
use percent_encoding::{percent_decode_str, utf8_percent_encode, AsciiSet, NON_ALPHANUMERIC};
use reqwest::{header::HeaderValue, Request, Response, Url};
use sha2::{Digest, Sha256};
use crate::model::{Credentials, Region};

/// What SigV4 leaves as it is: RFC 3986's unreserved characters. Everything else is `%XX`.
pub(crate) const STRICT: &AsciiSet = &NON_ALPHANUMERIC.remove(b'-').remove(b'.').remove(b'_').remove(b'~');

/// Where the bucket goes on the wire. The description always writes it in the path.
/// `VirtualHosted`: `{bucket}.s3.{region}.amazonaws.com/{key}`, AWS's own (a name with a dot
/// stays in the path, where TLS still matches). `Path`: as written, for MinIO, Ceph, most copies.
#[derive(Clone, Copy, PartialEq, Eq)]
pub enum Addressing { VirtualHosted, Path }

/// Credentials, a region, and where the bucket goes. Cheap to clone: keep one beside the
/// client, for pre-signed URLs.
#[derive(Clone)]
pub struct Signer { credentials: Credentials, region: Region, addressing: Addressing }

/// The transport: signs each try as it leaves, then hands it to `below` (Concord's HTTP client,
/// one of yours, or a fake). Each try gets its own date and its own signature.
pub struct SigV4<T> { pub(crate) signer: Signer, pub(crate) below: T }

impl<T: Transport> Transport for SigV4<T> {
    async fn send(&self, mut request: Request, info: TryInfo) -> Result<Response, TransportError> {
        self.signer.sign(&mut request, Utc::now()).map_err(|e| TransportError::blocked().with_source(e))?;
        self.below.send(request, info).await
    }
}

impl Signer {
    pub fn new(region: Region, credentials: Credentials) -> Signer {
        Signer { credentials, region, addressing: Addressing::VirtualHosted }
    }
    pub fn addressing(self, addressing: Addressing) -> Signer { Signer { addressing, ..self } }
    pub fn region(&self) -> &Region { &self.region }

    /// Signs `request` in its headers. The path and the query are rewritten to the form that
    /// is signed, so that what is sent is what is signed.
    fn sign(&self, request: &mut Request, now: DateTime<Utc>) -> Result<(), SignError> {
        if self.addressing == Addressing::VirtualHosted { move_bucket_to_host(request.url_mut())?; }
        canonicalize(request.url_mut());
        let payload = match request.body() {
            None => hex::encode(Sha256::digest(b"")),
            Some(body) => match body.as_bytes() {
                Some(bytes) => hex::encode(Sha256::digest(bytes)),
                None => "UNSIGNED-PAYLOAD".to_owned(),         // a stream: its bytes are not here yet
            },
        };
        let stamp = now.format("%Y%m%dT%H%M%SZ").to_string();
        let headers = request.headers_mut();
        headers.insert("x-amz-date", HeaderValue::from_str(&stamp)?);
        headers.insert("x-amz-content-sha256", HeaderValue::from_str(&payload)?);
        if let Some(token) = &self.credentials.session_token {
            headers.insert("x-amz-security-token", sensitive(&token.expose())?);
        }
        let (signed, lines) = canonical_headers(request);
        let url = request.url();
        let canonical = format!("{}\n{}\n{}\n{lines}\n{signed}\n{payload}",
                                request.method(), url.path(), url.query().unwrap_or(""));
        let authorization = format!(
            "AWS4-HMAC-SHA256 Credential={}/{}, SignedHeaders={signed}, Signature={}",
            self.credentials.access_key_id, self.scope(&stamp), self.signature(&stamp, &canonical),
        );
        request.headers_mut().insert("authorization", sensitive(&authorization)?);
        Ok(())
    }

    /// A URL that does what `request` would do, for anyone who holds it, for `expires_in`
    /// (7 days at most). Only the method, the URL and `host` are signed: the headers of the
    /// request are not carried, and its body is dropped.
    pub fn presign(&self, request: Request, expires_in: Duration) -> Result<Url, SignError> {
        if expires_in > Duration::from_secs(7 * 24 * 3600) { return Err(SignError::TooLong); }
        let (method, mut url) = (request.method().clone(), request.url().clone());
        if self.addressing == Addressing::VirtualHosted { move_bucket_to_host(&mut url)?; }
        let stamp = Utc::now().format("%Y%m%dT%H%M%SZ").to_string();
        {
            let mut query = url.query_pairs_mut();
            query.append_pair("X-Amz-Algorithm", "AWS4-HMAC-SHA256")
                .append_pair("X-Amz-Credential", &format!("{}/{}", self.credentials.access_key_id, self.scope(&stamp)))
                .append_pair("X-Amz-Date", &stamp)
                .append_pair("X-Amz-Expires", &expires_in.as_secs().to_string())
                .append_pair("X-Amz-SignedHeaders", "host");
            if let Some(token) = &self.credentials.session_token {
                query.append_pair("X-Amz-Security-Token", &token.expose());
            }
        }
        canonicalize(&mut url);
        let canonical = format!("{method}\n{}\n{}\nhost:{}\n\nhost\nUNSIGNED-PAYLOAD",
                                url.path(), url.query().unwrap_or(""), host(&url));
        let signature = self.signature(&stamp, &canonical);
        url.query_pairs_mut().append_pair("X-Amz-Signature", &signature);
        Ok(url)
    }

    fn scope(&self, stamp: &str) -> String { format!("{}/{}/s3/aws4_request", &stamp[..8], self.region.as_str()) }

    fn signature(&self, stamp: &str, canonical: &str) -> String {
        let hashed = hex::encode(Sha256::digest(canonical));
        let to_sign = format!("AWS4-HMAC-SHA256\n{stamp}\n{}\n{hashed}", self.scope(stamp));
        let key = [&stamp[..8], self.region.as_str(), "s3", "aws4_request"].into_iter().fold(
            format!("AWS4{}", self.credentials.secret_access_key.expose()).into_bytes(),
            |key, part| hmac(&key, part.as_bytes()),
        );
        hex::encode(hmac(&key, to_sign.as_bytes()))
    }
}

/// `https://s3.eu-west-1.amazonaws.com/photos/a/b.jpg` becomes
/// `https://photos.s3.eu-west-1.amazonaws.com/a/b.jpg`. No bucket (ListBuckets), or a bucket
/// with a dot: left as it is. Each try gets a fresh copy of the request, so this runs once on it.
fn move_bucket_to_host(url: &mut Url) -> Result<(), SignError> {
    let path = url.path().trim_start_matches('/').to_owned();
    let (bucket, rest) = path.split_once('/').unwrap_or((path.as_str(), ""));
    if bucket.is_empty() || bucket.contains('.') { return Ok(()); }
    let host = format!("{bucket}.{}", url.host_str().unwrap_or_default());
    url.set_host(Some(&host)).map_err(SignError::Host)?;
    url.set_path(&format!("/{rest}"));
    Ok(())
}

/// The path and the query in the one form that SigV4 signs: each piece decoded, then encoded
/// with `STRICT`; the pairs sorted. (Concord writes a space as `%20` and never a bare `+`.)
fn canonicalize(url: &mut Url) {
    let path: Vec<String> = url.path().split('/')
        .map(|piece| utf8_percent_encode(&percent_decode_str(piece).decode_utf8_lossy(), STRICT).to_string())
        .collect();
    let mut pairs: Vec<(String, String)> = url.query_pairs()
        .map(|(k, v)| (utf8_percent_encode(&k, STRICT).to_string(), utf8_percent_encode(&v, STRICT).to_string()))
        .collect();
    pairs.sort();
    let query: Vec<String> = pairs.into_iter().map(|(k, v)| format!("{k}={v}")).collect();
    url.set_path(&path.join("/"));
    url.set_query(if query.is_empty() { None } else { Some(&query.join("&")) });
}

/// `host` as SigV4 signs it: with the port when the URL names one.
fn host(url: &Url) -> String {
    let name = url.host_str().unwrap_or_default();
    url.port().map_or_else(|| name.to_owned(), |port| format!("{name}:{port}"))
}

/// The signed headers: `host`, `content-type`, `content-md5`, and every `x-amz-*`.
fn canonical_headers(request: &Request) -> (String, String) {
    let mut pairs = vec![("host".to_owned(), host(request.url()))];
    for (name, value) in request.headers() {
        let name = name.as_str();                          // lower case already
        if name.starts_with("x-amz-") || name == "content-type" || name == "content-md5" {
            let value = value.to_str().unwrap_or_default().split_whitespace().collect::<Vec<_>>().join(" ");
            pairs.push((name.to_owned(), value));
        }
    }
    pairs.sort();
    let signed = pairs.iter().map(|(name, _)| name.as_str()).collect::<Vec<_>>().join(";");
    let lines = pairs.iter().map(|(name, value)| format!("{name}:{value}\n")).collect();
    (signed, lines)
}

fn hmac(key: &[u8], data: &[u8]) -> Vec<u8> {
    let mut mac = Hmac::<Sha256>::new_from_slice(key).expect("HMAC takes a key of any length");
    mac.update(data);
    mac.finalize().into_bytes().to_vec()
}

fn sensitive(text: &str) -> Result<HeaderValue, SignError> {
    let mut value = HeaderValue::from_str(text)?;
    value.set_sensitive(true);
    Ok(value)
}

#[derive(Debug, thiserror::Error)]
pub enum SignError {
    #[error("a header value that cannot be sent")] Header(#[from] reqwest::header::InvalidHeaderValue),
    #[error("the bucket name cannot be put in the host")] Host(url::ParseError),
    #[error("a pre-signed URL lives 7 days at most")] TooLong,
}
```

```rust
impl S3 {
    /// **The way in, for Amazon S3.** Every request signed by `signer`, the bucket in the host.
    pub fn signed(signer: Signer) -> reqwest::Result<S3> {
        Ok(S3::signed_with(signer, concord::Http::with(|builder| builder)?))
    }

    /// The way in for a store that copies S3 (MinIO, Ceph, R2, …) at `endpoint`: every
    /// request signed, the bucket in the path.
    pub fn compatible(endpoint: concord::Base, signer: Signer) -> reqwest::Result<S3> {
        Ok(S3::signed(signer.addressing(Addressing::Path))?.with_base(endpoint))
    }

    /// `signed`, over a transport of yours: your own HTTP client, or a fake in a test.
    /// (`S3::signed(…)?.with_transport(…)` would replace the signer too.)
    pub fn signed_with(signer: Signer, below: impl Transport) -> S3 {
        S3::new(signer.region().clone()).with_transport(SigV4 { signer, below })
    }
}

/// User metadata: `x-amz-meta-{name}`, a name that the caller chooses, so not in the
/// description. The value must be ASCII (S3 asks it; Concord refuses anything else anyway).
macro_rules! user_metadata {
    ($($call:ty),*) => {$(
        impl $call {
            pub fn meta(self, name: &str, value: &str) -> Result<Self, InvalidMeta> {
                let name = HeaderName::try_from(format!("x-amz-meta-{}", name.to_ascii_lowercase()))
                    .map_err(|_| InvalidMeta)?;
                let value = HeaderValue::from_str(value).map_err(|_| InvalidMeta)?;
                Ok(self.edit_request(move |request| { request.headers_mut().insert(name.clone(), value.clone()); }))
            }
        }
    )*};
}
user_metadata!(s3::bucket::object::Put, s3::bucket::object::CopyFrom, s3::bucket::object::CreateMultipartUpload);

/// The retry that S3 asks for, beyond the standard table: `InternalError`, as a `500` or
/// inside a `200` (CompleteMultipartUpload, CopyObject). `SlowDown` is a `503`, already in it.
pub fn retry() -> concord::Retry<S3Problem> {
    concord::Retry::standard().when(|tried| {
        let internal = tried.status().is_some_and(|s| s == 500)
            || tried.problem().is_some_and(|p| p.code == "InternalError");
        internal.then_some(Duration::ZERO)                  // the standard wait still applies (unsure)
    })
}
```

```rust
async fn upload_large(bucket: &s3::Bucket, key: &str, path: &Path) -> anyhow::Result<ETag> {

    let upload = object.multipart(&started.upload_id);
    let sent = async {
        let mut file = tokio::fs::File::open(path).await?;
        let mut parts = Vec::new();
        for number in 1u16.. {
            let chunk = read_up_to(&mut file, 16 * 1024 * 1024).await?;      // a `Bytes`; empty at the end
            if chunk.is_empty() { break; }
            let part = upload.upload_part(number, concord::Body::from(chunk))  // (unsure: no constructor in the guide)

                .await?;

    }.await;
    if sent.is_err() {
        upload.abort().await.ok();                 // an upload left open is billed until it is aborted
    }
    Ok(sent?.etag)
}
```
