//! Baseline recording harness (v2.1). Executes every Riot endpoint and the
//! Catalog contract endpoints through the real generated client with a
//! recording limiter that denies admission after capturing the complete
//! protected rate plan. No request reaches a network: limiter denial happens
//! before native handoff.
//!
//! Set `CONCORD_BASELINE_RECORDING_OUT=/path/file.json` to write the recording.
#![forbid(unsafe_code)]
#![cfg(all(feature = "json", feature = "rate-limit", feature = "default-tls"))]

use std::fmt::Write as _;
use std::future::Future;
use std::sync::{Arc, Mutex};

use concord::ApiError;
use concord::SecretString;
use concord::observation::SanitizedHeaderValue;
use concord::rate_limit::{
    BoxFuture, RateLimitError, RateLimitErrorKind, RateLimitPermit, RateLimitRequest, RateLimiter,
};
use concord_examples::riot::api::RiotApi;
use concord_examples::riot::domain::*;

#[derive(Default)]
struct Recorder {
    entries: Mutex<Vec<String>>,
}

struct RecordingLimiter(Arc<Recorder>);

impl RateLimiter for RecordingLimiter {
    fn acquire<'a>(
        &'a self,
        request: RateLimitRequest<'a>,
    ) -> BoxFuture<'a, Result<RateLimitPermit, RateLimitError>> {
        let mut out = String::new();
        write!(
            out,
            "{{\"endpoint\":{:?},\"method\":{:?},\"logical_origin\":{},\"policies\":[",
            request.endpoint,
            request.method.as_str(),
            match request.logical_origin {
                Some(origin) => format!("{origin:?}"),
                None => "null".to_owned(),
            }
        )
        .unwrap();
        for (index, policy) in request.plan.policies().iter().enumerate() {
            if index > 0 {
                out.push(',');
            }
            write!(
                out,
                "{{\"id\":{:?},\"cost\":{},\"windows\":[",
                policy.id().as_str(),
                policy.cost()
            )
            .unwrap();
            for (w, window) in policy.windows().iter().enumerate() {
                if w > 0 {
                    out.push(',');
                }
                write!(
                    out,
                    "[{},{}]",
                    window.capacity(),
                    window.duration().as_millis()
                )
                .unwrap();
            }
            out.push_str("],\"key\":[");
            for (p, part) in policy.key().parts().iter().enumerate() {
                if p > 0 {
                    out.push(',');
                }
                let hex = part
                    .value()
                    .iter()
                    .map(|byte| format!("{byte:02x}"))
                    .collect::<String>();
                write!(out, "[{:?},{:?}]", part.name(), hex).unwrap();
            }
            out.push_str("]}");
        }
        out.push_str("]}");
        self.0.entries.lock().unwrap().push(out);
        Box::pin(async { Err(RateLimitError::new(RateLimitErrorKind::Acquire)) })
    }
}

fn riot(recorder: &Arc<Recorder>) -> RiotApi {
    let limiter = RecordingLimiter(Arc::clone(recorder));
    RiotApi::builder()
        .api_key(SecretString::try_new("riot-baseline-sentinel").unwrap())
        .configure_runtime(|runtime| runtime.rate_limiter(Arc::new(limiter)))
        .build()
        .expect("baseline Riot client")
}

fn header_line(plan: &concord::endpoint::RequestPlan) -> String {
    let mut headers = String::from("[");
    for (index, (name, value)) in plan.public_headers().iter().enumerate() {
        if index > 0 {
            headers.push(',');
        }
        let value = match value {
            SanitizedHeaderValue::Text(text) => format!("{text:?}"),
            SanitizedHeaderValue::Redacted => "\"<redacted>\"".to_owned(),
            SanitizedHeaderValue::NonUtf8 => "\"<non-utf8>\"".to_owned(),
        };
        write!(headers, "[{:?},{value}]", name.as_str()).unwrap();
    }
    headers.push(']');
    headers
}

async fn record<F, T>(
    label: &str,
    recorder: &Arc<Recorder>,
    plan: Result<concord::endpoint::RequestPlan, ApiError>,
    execution: F,
    lines: &mut Vec<String>,
) where
    F: Future<Output = Result<T, ApiError>>,
{
    let plan = plan.expect("baseline plan");
    let before = recorder.entries.lock().unwrap().len();
    let result = execution.await;
    assert!(result.is_err(), "{label}: recording limiter must deny");
    let entries = recorder.entries.lock().unwrap();
    assert_eq!(entries.len(), before + 1, "{label}: exactly one admission");
    lines.push(format!(
        "{{\"label\":{label:?},\"endpoint_id\":{:?},\"method\":{:?},\"observed_url\":{:?},\"timeout\":{:?},\"headers\":{},\"limiter\":{}}}",
        plan.endpoint_id(),
        plan.method().as_str(),
        plan.observed_url(),
        format!("{:?}", plan.timeout_selection()),
        header_line(&plan),
        entries[before]
    ));
}

macro_rules! rec {
    ($lines:ident, $recorder:ident, $client:ident, $label:expr, $unbound:expr, $bound:expr) => {{
        let endpoint = $unbound;
        let plan = $client.plan(&endpoint);
        record($label, &$recorder, plan, $bound, &mut $lines).await;
    }};
}

#[tokio::test]
async fn baseline_riot_recording() {
    let recorder = Arc::new(Recorder::default());
    let client = riot(&recorder);
    let mut lines = Vec::new();
    for platform in [PlatformRoute::Na1, PlatformRoute::Euw1] {
        let p = format!("{platform:?}");
        let u = || RiotApi::unbound().platform(platform);
        let b = || client.platform(platform);
        macro_rules! both {
            ($name:literal, $($path:tt)+) => {
                rec!(lines, recorder, client, &format!("{}:{}", p, $name), u().$($path)+, b().$($path)+.into_future())
            };
        }
        use std::future::IntoFuture;
        both!("masteries_by_puuid", champion_mastery_v4().masteries_by_puuid("puuid-a".to_owned()));
        both!("mastery_by_champion", champion_mastery_v4().mastery_by_champion("puuid-a".to_owned(), 7));
        both!("top_masteries_by_puuid", champion_mastery_v4().top_masteries_by_puuid("puuid-a".to_owned()).set_count(Some(3)));
        both!("mastery_score_by_puuid", champion_mastery_v4().mastery_score_by_puuid("puuid-a".to_owned()));
        both!("champion_rotations", champion_v3().champion_rotations());
        both!("clash_players_by_puuid", clash_v1().clash_players_by_puuid("puuid-a".to_owned()));
        both!("clash_team", clash_v1().clash_team("team-a".to_owned()));
        both!("clash_tournament_by_team", clash_v1().clash_tournament_by_team("team-a".to_owned()));
        both!("clash_tournament", clash_v1().clash_tournament(11));
        both!("clash_tournaments", clash_v1().clash_tournaments());
        both!("challenger_by_queue", league_v4().challenger_by_queue(LeagueQueue::RankedSolo5x5));
        both!("grandmaster_by_queue", league_v4().grandmaster_by_queue(LeagueQueue::RankedSolo5x5));
        both!("master_by_queue", league_v4().master_by_queue(LeagueQueue::RankedFlexSr));
        both!("league_entries", league_v4().league_entries(LeagueQueue::RankedSolo5x5, LeagueTier::Gold, LeagueDivision::Ii));
        both!("league_entries_by_puuid", league_v4().league_entries_by_puuid("puuid-a".to_owned()));
        both!("league_by_id", league_v4().league_by_id("league-a".to_owned()));
        both!("league_exp_entries", league_exp_v4().league_exp_entries(LeagueQueue::RankedSolo5x5, LeagueTier::Gold, LeagueDivision::I).set_page(2));
        both!("challenge_configs", challenges_v1().challenge_configs());
        both!("challenge_percentiles", challenges_v1().challenge_percentiles());
        both!("challenge_config", challenges_v1().challenge_config(5));
        both!("challenge_leaderboard", challenges_v1().challenge_leaderboard(5, ChallengeLevel::Master).set_limit(Some(10)));
        both!("challenge_percentiles_by_id", challenges_v1().challenge_percentiles_by_id(5));
        both!("challenge_player_data", challenges_v1().challenge_player_data("puuid-a".to_owned()));
        both!("platform_data", status_v4().platform_data());
        both!("active_game_by_puuid", spectator_v5().active_game_by_puuid("puuid-a".to_owned()));
        both!("summoner_by_puuid", summoner_v4().summoner_by_puuid("puuid-a".to_owned()));
        both!("summoner_by_id", summoner_v4().summoner_by_id("summoner-a".to_owned()));
    }
    for regional in [RegionalRoute::Americas, RegionalRoute::Europe] {
        let p = format!("{regional:?}");
        let u = || RiotApi::unbound().regional(regional);
        let b = || client.regional(regional);
        macro_rules! both {
            ($name:literal, $($path:tt)+) => {
                rec!(lines, recorder, client, &format!("{}:{}", p, $name), u().$($path)+, b().$($path)+.into_future())
            };
        }
        use std::future::IntoFuture;
        both!("account_by_riot_id", account_v1().account_by_riot_id("game".to_owned(), "tag".to_owned()));
        both!("account_by_puuid", account_v1().account_by_puuid("puuid-a".to_owned()));
        both!("active_region", account_v1().active_region(RiotGame::LeagueOfLegends, "puuid-a".to_owned()));
        both!("match_ids_by_puuid", match_v5().match_ids_by_puuid("puuid-a".to_owned()).set_start_time(Some(100)).set_end_time(Some(200)).set_queue(Some(420)).set_match_type(Some(MatchType::Ranked)));
        both!("replays_by_puuid", match_v5().replays_by_puuid("puuid-a".to_owned()));
        both!("match_by_id", match_v5().match_by_id("match-a".to_owned()));
        both!("timeline_by_id", match_v5().timeline_by_id("match-a".to_owned()));
        both!("create_stub_codes", tournament_stub_v5().create_stub_codes(9, TournamentCodeParameters::default()).set_count(Some(2)));
        both!("stub_code", tournament_stub_v5().stub_code("code-a".to_owned()));
        both!("stub_lobby_events", tournament_stub_v5().stub_lobby_events("code-a".to_owned()));
        both!("register_stub_provider", tournament_stub_v5().register_stub_provider(ProviderRegistrationParameters::default()));
        both!("register_stub_tournament", tournament_stub_v5().register_stub_tournament(TournamentRegistrationParameters::default()));
        both!("create_tournament_codes", tournament_v5().create_tournament_codes(9, TournamentCodeParameters::default()).set_count(Some(2)));
        both!("tournament_code", tournament_v5().tournament_code("code-a".to_owned()));
        both!("update_tournament_code", tournament_v5().update_tournament_code("code-a".to_owned(), TournamentCodeUpdateParameters::default()));
        both!("tournament_game_by_code", tournament_v5().tournament_game_by_code("code-a".to_owned()));
        both!("tournament_lobby_events", tournament_v5().tournament_lobby_events("code-a".to_owned()));
        both!("register_tournament_provider", tournament_v5().register_tournament_provider(ProviderRegistrationParameters::default()));
        both!("register_tournament", tournament_v5().register_tournament(TournamentRegistrationParameters::default()));
    }
    assert_eq!(lines.len(), 92, "46 endpoints x 2 origins");
    if let Ok(path) = std::env::var("CONCORD_BASELINE_RECORDING_OUT") {
        std::fs::write(path, format!("[\n{}\n]\n", lines.join(",\n"))).unwrap();
    }
}

#[tokio::test]
async fn baseline_catalog_recording() {
    use concord_examples::contracts::catalog_contract::{CatalogApi, SessionToken, UpdateInput};
    let recorder = Arc::new(Recorder::default());
    let mut lines = Vec::new();
    for tenant in ["tenant-a", "tenant-b"] {
        let limiter = RecordingLimiter(Arc::clone(&recorder));
        let client = CatalogApi::builder()
            .set_tenant(tenant.to_owned())
            .bootstrap_key(SecretString::try_new("catalog-bootstrap-sentinel").unwrap())
            .configure_runtime(|runtime| runtime.rate_limiter(Arc::new(limiter)))
            .build()
            .expect("baseline Catalog client");
        let token: SessionToken = serde_json::from_str("\"session-sentinel\"").unwrap();
        client.credentials().session().set(token).await.expect("install session");
        for region in ["eu", "us"] {
            let label = format!("{tenant}:{region}");
            use std::future::IntoFuture;
            rec!(lines, recorder, client, &format!("{label}:item_summaries"),
                CatalogApi::unbound().regional(region.to_owned()).item_summaries(vec!["t1".to_owned(), "t2".to_owned()]),
                client.regional(region.to_owned()).item_summaries(vec!["t1".to_owned(), "t2".to_owned()]).into_future());
            rec!(lines, recorder, client, &format!("{label}:item"),
                CatalogApi::unbound().regional(region.to_owned()).item("item-1".to_owned()),
                client.regional(region.to_owned()).item("item-1".to_owned()).into_future());
            rec!(lines, recorder, client, &format!("{label}:update_item"),
                CatalogApi::unbound().regional(region.to_owned()).update_item("item-1".to_owned(), "cmd-1".to_owned(), UpdateInput { title: "t".to_owned() }),
                client.regional(region.to_owned()).update_item("item-1".to_owned(), "cmd-1".to_owned(), UpdateInput { title: "t".to_owned() }).into_future());
        }
    }
    assert_eq!(lines.len(), 12);
    if let Ok(path) = std::env::var("CONCORD_BASELINE_CATALOG_RECORDING_OUT") {
        std::fs::write(path, format!("[\n{}\n]\n", lines.join(",\n"))).unwrap();
    }
}
